# 🎯 小红书笔记完整内容提取 - 最简单方案

## 📌 方案一:使用自动提取脚本(最推荐)

### 步骤:

1. **在你已登录的浏览器中,打开小红书**
   - 访问: https://www.xiaohongshu.com
   - 确保已经登录

2. **打开任意一篇你的笔记**
   - 点击进入你想提取的第一篇笔记

3. **打开浏览器控制台**
   - 按 `F12` 键
   - 切换到 "Console(控制台)" 标签

4. **复制并运行以下脚本:**

```javascript
(async function() {
    const response = await fetch('data/notes.json');
    const notes = await response.json();
    const notesList = notes.map(n => ({url: n.url, title: n.title || '无标题'}));

    let extractedNotes = [];
    let currentIndex = 0;

    function extractCurrentPage() {
        console.clear();
        console.log('📝 正在提取...\n');

        const titleEl = document.querySelector('[class*="title"], h1, h2');
        const title = titleEl ? titleEl.innerText.trim() : document.title.split('-')[0].trim();

        let content = '';
        const selectors = [
            '[class*="note-text"]',
            '[class*="desc-text"]',
            '[class*="content-text"]',
            'section[class*="note"] div[class*="text"]',
            'article[class*="note"]',
            'div[class*="rich-text"]'
        ];

        for (const selector of selectors) {
            const elements = document.querySelectorAll(selector);
            if (elements.length > 0) {
                const texts = Array.from(elements).map(el => {
                    const cloned = el.cloneNode(true);
                    cloned.querySelectorAll('script, style, nav, footer, header').forEach(s => s.remove());
                    return cloned.innerText.trim();
                }).filter(text => text && text.length > 20).join('\n\n');

                if (texts.length > content.length) content = texts;
            }
        }

        const imgElements = document.querySelectorAll('img');
        const images = Array.from(imgElements).map(img => img.src)
            .filter(src => src && (src.includes('sns-webpic') || src.includes('xhscdn'))).slice(0, 9);

        const result = {url: window.location.href, title: title, content: content, images: images, extractedAt: new Date().toISOString()};
        extractedNotes.push(result);

        console.log('✅ 完成! 进度: ' + extractedNotes.length + '/' + notesList.length);
        console.log('📝 ' + title + ' (' + content.length + ' 字符)\n');

        return result;
    }

    function showPrompt() {
        console.clear();
        console.log('\n═══════════════════════════════════════');
        console.log('📝 小红书笔记提取工具');
        console.log('═══════════════════════════════════════\n');
        console.log('📊 进度: ' + extractedNotes.length + '/' + notesList.length + '\n');

        if (currentIndex < notesList.length) {
            const note = notesList[currentIndex];
            console.log('─────────────────────────────────────\n');
            console.log('📋 下一步:\n');
            console.log('1. 访问: ' + note.url);
            console.log('2. 页面加载后,输入: extract()\n');
            console.log('═══════════════════════════════════════\n');
        } else {
            console.log('🎉 全部完成!\n');
            console.log('输入 showJSON() 查看结果\n');
        }
    }

    window.extract = function() {
        extractCurrentPage();
        currentIndex++;
        showPrompt();
    };

    window.showJSON = function() {
        console.log('\n📋 提取的数据:\n');
        console.log(JSON.stringify(extractedNotes, null, 2));
    };

    window.saveData = function() {
        localStorage.setItem('xhs_extracted_notes', JSON.stringify(extractedNotes));
        console.log('\n✅ 已保存到浏览器存储\n');
    };

    console.log('✅ 工具已加载!\n命令: extract() - 提取当前页面\n      showJSON() - 显示数据\n      saveData() - 保存数据\n');

    showPrompt();
})();
```

5. **开始提取:**
   - 脚本会显示第一篇笔记的URL
   - 访问该URL
   - 在控制台输入: `extract()`
   - 等待提取完成
   - 继续下一篇...

6. **查看结果:**
   - 提取完成后,在控制台输入: `showJSON()`
   - 复制输出的JSON数据

## 💡 提示

1. **登录状态**: 确保在小红书网页已登录

## 🆘 常见问题

**Q: 提取的内容为空?**
A: 可能页面未加载完成,等待几秒后再运行 `extract()`

**Q: 如何暂停?**
A: 直接关闭浏览器即可,下次使用 `loadData()` 恢复进度

**Q: 如何查看已提取了多少?**
A: 输入 `showJSON()` 查看所有已提取的数据
