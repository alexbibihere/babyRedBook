# 🎯 完整实现方案:小红书收藏和点赞笔记爬取与整理

## 📋 需求分析

### 目标
1. 爬取个人账号的**收藏**和**点赞**的笔记
2. 提取每篇笔记的**文字内容**
3. 识别并处理**图片内容**(OCR识别)
4. 将所有内容整理成**一篇完整文档**

### 数据来源
- ✅ 收藏的笔记
- ✅ 点赞的笔记
- ✅ 每篇包含:文字、图片

(async function() {
    const notes = [];

    // 自动滚动加载
    async function autoScroll() {
        while (true) {
            // 提取当前页面的笔记

**实现代码**:
```javascript
// 使用PaddleOCR识别图片
const { PaddleOCR } = require('paddleocr');

const ocr = new PaddleOCR();

async function extractTextFromImage(imageUrl) {
    // 1. 下载图片
    const imagePath = await downloadImage(imageUrl);

    // 2. OCR识别
    const result = await ocr.ocr(imagePath);

    // 3. 提取文字
    const text = result.map(r => r.text).join('\n');

    return text;
}
```

#### 阶段4:内容整理

**处理逻辑**:
```javascript
async function processNotes(notes) {
    const results = [];

    for (const note of notes) {
        // 1. 提取文字内容
        const textContent = note.content || '';

        // 2. 识别图片文字
        let imageTexts = [];
        for (const imageUrl of note.images) {
            const text = await extractTextFromImage(imageUrl);
            imageTexts.push(text);
        }

        // 3. 合并内容
        const fullContent = [
            `# ${note.title}`,
            `**发布时间**: ${note.createTime}`,
            `**点赞**: ${note.likes}`,
            `**链接**: ${note.url}`,
            '',
            '## 正文',
            textContent,
            '',
            '## 图片内容',
            ...imageTexts.map((text, i) => [
                `### 图片${i + 1}`,
                text,
                ''
            ]).flat()
        ].join('\n');

        results.push({
            ...note,
            fullContent: fullContent,
            imageTexts: imageTexts
        });
    }

    return results;
}
```

// 生成Markdown文档
function generateMarkdown(notes) {
    let markdown = `# 我的小红书收藏与点赞笔记合集\n\n`;
    markdown += `> 生成时间: ${new Date().toLocaleString()}\n`;
    markdown += `> 总计: ${notes.length} 篇笔记\n\n`;

    markdown += `---\n\n`;

    // 目录
    markdown += `## 目录\n\n`;
    notes.forEach((note, i) => {
        markdown += `${i + 1}. [${note.title}](#${note.id})\n`;
    });

    markdown += `\n---\n\n`;

    // 正文
    notes.forEach((note, i) => {
        markdown += `## ${i + 1}. ${note.title}\n\n`;
        markdown += note.fullContent;
        markdown += `\n---\n\n`;
    });

    return markdown;
}

// 保存文档
fs.writeFileSync('我的小红书笔记合集.md', markdown);

## 💻 核心代码实现

### 完整处理脚本

```javascript
import fs from 'fs';
import PaddleOCR from 'paddleocr';
import axios from 'axios';

class NoteProcessor {
    constructor() {
        this.ocr = new PaddleOCR();
        this.notesData = JSON.parse(fs.readFileSync('notes.json', 'utf-8'));
    }

    // 下载图片
    async downloadImage(url, filepath) {
        const response = await axios({
            method: 'GET',
            url: url,
            responseType: 'stream'
        });

        const writer = fs.createWriteStream(filepath);
        response.data.pipe(writer);

        return new Promise((resolve, reject) => {
            writer.on('finish', resolve);
            writer.on('error', reject);
        });
    }

    // OCR识别
    async extractImageText(imageUrl) {
        const filename = `temp_${Date.now()}.jpg`;
        await this.downloadImage(imageUrl, filename);

        const result = await this.ocr.ocr(filename);
        const text = result.map(r => r.text).join('\n');

        fs.unlinkSync(filename); // 删除临时文件
        return text;
    }

    // 处理单篇笔记
    async processNote(note) {
        console.log(`处理: ${note.title}`);

        const imageTexts = [];
        for (let i = 0; i < note.images.length; i++) {
            console.log(`  识别图片 ${i + 1}/${note.images.length}`);
            const text = await this.extractImageText(note.images[i]);
            imageTexts.push(text);
        }

        return {
            ...note,
            imageTexts: imageTexts
        };
    }

    // 处理所有笔记
    async processAll() {
        const results = [];

        for (let i = 0; i < this.notesData.length; i++) {
            const note = this.notesData[i];
            const processed = await this.processNote(note);
            results.push(processed);

            // 每10篇保存一次
            if ((i + 1) % 10 === 0) {
                fs.writeFileSync(
                    `notes-progress-${i + 1}.json`,
                    JSON.stringify(results, null, 2)
                );
            }
        }

        return results;
    }

    // 生成文档
    generateDocument(processedNotes) {
        let markdown = `# 我的小红书笔记合集\n\n`;
        markdown += `生成时间: ${new Date().toLocaleString()}\n`;
        markdown += `总计: ${processedNotes.length} 篇\n\n---\n\n`;

        processedNotes.forEach((note, i) => {
            markdown += `## ${i + 1}. ${note.title}\n\n`;
            markdown += `**时间**: ${note.createTime}\n`;
            markdown += `**链接**: ${note.url}\n\n`;

            // 正文
            markdown += `### 正文内容\n\n${note.content}\n\n`;

            // 图片文字
            if (note.imageTexts && note.imageTexts.length > 0) {
                markdown += `### 图片识别内容\n\n`;
                note.imageTexts.forEach((text, j) => {
                    markdown += `#### 图片${j + 1}\n\n${text}\n\n`;
                });
            }

            markdown += `---\n\n`;
        });

        fs.writeFileSync('我的小红书笔记合集-完整版.md', markdown);
        console.log('✅ 文档生成完成!');
    }
}

// 使用
const processor = new NoteProcessor();
const processed = await processor.processAll();
processor.generateDocument(processed);
```
